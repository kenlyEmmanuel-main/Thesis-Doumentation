% ==============================================================================
% Geometry & Margin Settings (ADNU ECE/CpE Section A & C)
% Top: 1.0", Bottom: 1.0", Right: 1.0", Left: 1.25"
% Pagination: 0.5" from top/bottom
% ==============================================================================
\usepackage[
    top=1.0in,
    bottom=1.0in,
    right=1.0in,
    left=1.25in,
    headheight=14.5pt,
    headsep=0.5in,
    footskip=0.5in
]{geometry}

% ==============================================================================
% Typography & Font Styling (ADNU ECE/CpE Section B)
% newtx supplies Times-like text and math, not the Times New Roman font.
% ==============================================================================
\usepackage{newtxtext, newtxmath}

% ==============================================================================
% Line Spacing & Paragraph Formatting (ADNU ECE/CpE Section F)
% Body double spaced, 0pt before/after, 1.50cm first line indent
% ==============================================================================
\usepackage{setspace}
\usepackage{indentfirst}
\doublespacing
\setlength{\parindent}{1.50cm}
\setlength{\parskip}{0pt}

% ==============================================================================
% Header & Footer Pagination (ADNU ECE/CpE Section C)
% Titles of major sections: page number bottom center, 0.5" from bottom
% Pages with no major section titles: page number upper right corner, 0.5" from top
% ==============================================================================
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[R]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% Redefine plain style used automatically by \chapter
\fancypagestyle{plain}{
    \fancyhf{}
    \fancyfoot[C]{\thepage}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
}

% ==============================================================================
% Heading Formats & Hierarchy (ADNU ECE/CpE Section F)
% Chapter: "Chapter X" (bold, title case) \\ "TITLE" (bold, all caps, centered)
% ==============================================================================
\usepackage{titlesec}

\titleformat{\chapter}[display]
    {\normalfont\fontsize{12pt}{15pt}\bfseries\centering}
    {Chapter \thechapter}
    {1\baselineskip}
    {\MakeUppercase}
\titlespacing*{\chapter}{0pt}{0pt}{1\baselineskip}

\titleformat{\section}
    {\normalfont\fontsize{12pt}{15pt}\bfseries}
    {\thesection}
    {1em}
    {}
\titlespacing*{\section}{0pt}{30pt plus 4pt minus 2pt}{0pt}

\titleformat{\subsection}
    {\normalfont\fontsize{12pt}{15pt}\bfseries\raggedright}
    {\thesubsection}
    {1em}
    {}
\titlespacing*{\subsection}{0pt}{30pt plus 4pt minus 2pt}{0pt}

\titleformat{\subsubsection}
    {\normalfont\fontsize{12pt}{15pt}\bfseries\raggedright}
    {\thesubsubsection}
    {1em}
    {}
\titlespacing*{\subsubsection}{0pt}{30pt plus 4pt minus 2pt}{0pt}

\setcounter{secnumdepth}{4}
\setcounter{tocdepth}{3}
\makeatletter
\renewcommand*\l@subsection{\@dottedtocline{2}{1.5em}{3.5em}}
\renewcommand*\l@subsubsection{\@dottedtocline{3}{5em}{4.5em}}
\makeatother
